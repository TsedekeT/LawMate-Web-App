// Function to validate password criteria
function validatePassword(password) {
    const minLength = password.length >= 8;
    const hasUppercase = /[A-Z]/.test(password);
    const hasLowercase = /[a-z]/.test(password);
    const hasNumber = /[0-9]/.test(password);
    const hasSpecialChar = /[@#$%^&*(),.?":{}|<>]/.test(password);
    return minLength && hasUppercase && hasLowercase && hasNumber && hasSpecialChar;
}

// Function to handle password validation for a specific form
function handlePasswordValidation(modalId) {
    const modal = document.querySelector(`#${modalId}`);
    const passwordInput = modal.querySelector('input[placeholder="Password"]');
    const passwordPolicy = modal.querySelector('#passwordPolicy');
    const submitButton = modal.querySelector('.modal-btn');

    // Display password policy if validation fails on submit
    submitButton.addEventListener('click', (e) => {
        const password = passwordInput.value;

        if (!validatePassword(password)) {
            e.preventDefault(); // Prevent form submission
            passwordPolicy.style.display = 'block';
            passwordPolicy.style.color = 'red'; // Optional: make it more prominent
        } else {
            passwordPolicy.style.display = 'none'; // Hide the message if validation passes
        }
    });

    // Optional: Hide the password policy when the user starts typing
    passwordInput.addEventListener('input', () => {
        passwordPolicy.style.display = 'none';
    });
}

// Initialize password validation for both modals
handlePasswordValidation('signup');
handlePasswordValidation('login');
